const express = require('express'); const cors = require('cors'); const app = express(); app.use(express.json()); app.use(cors()); // অ্যাপ ডাটা ও এডমিন তথ্য let appSettings = { gmailRate: 13, adminPhone: "01743815478", adminPass: "102050" }; let users = {}; let tasks = []; // ১. ফ্রন্টএন্ড হোম ও ইউজার ইন্টারফেস (Single HTML Output) app.get('/', (req, res) => { res.send(` Telegram Task Mini App

🌟 Premium Task Hub

টেলিগ্রাম মিনি অ্যাপ আয় মাধ্যম

কাজের তালিকা

Gmail Task Submit

আপনার ব্যালেন্স

0 BDT

টাকা উত্তোলন (Withdraw)

`); }); // ২. এডমিন প্যানেল রুট (/admin) app.get('/admin', (req, res) => { res.send(` Admin Panel

⚙️ Admin Panel

লগইন করুন

রেট পরিবর্তন করুন

পেন্ডিং টাস্ক লিস্ট

`); }); // API Endpoints app.get('/api/settings', (req, res) => res.json({ gmailRate: appSettings.gmailRate })); app.post('/api/admin/login', (req, res) => { const { phone, password } = req.body; if (phone === appSettings.adminPhone && password === appSettings.adminPass) { res.json({ success: true }); } else { res.status(401).json({ success: false }); } }); app.post('/api/admin/update-rate', (req, res) => { const { rate, phone, password } = req.body; if (phone === appSettings.adminPhone && password === appSettings.adminPass) { appSettings.gmailRate = parseFloat(rate); res.json({ success: true, message: "রেট আপডেট হয়েছে!" }); } else { res.status(401).json({ success: false, message: "অনুমোদিত নয়" }); } }); app.get('/api/user/:id', (req, res) => { if (!users[req.params.id]) users[req.params.id] = { balance: 0 }; res.json(users[req.params.id]); }); app.post('/api/submit-task', (req, res) => { const { userId, gmailName, password } = req.body; tasks.push({ id: Date.now(), userId, gmailName, password, rate: appSettings.gmailRate, status: 'Pending' }); res.json({ success: true, message: "টাস্ক সফলভাবে জমা হয়েছে!" }); }); app.get('/api/admin/tasks', (req, res) => res.json(tasks)); app.post('/api/admin/task-action', (req, res) => { const { taskId, action, phone, password } = req.body; if (phone !== appSettings.adminPhone || password !== appSettings.adminPass) return res.status(401).json({ message: "Unauthorized" }); const t = tasks.find(x => x.id == taskId); if (t && action === 'approve') { t.status = 'Approved'; if (!users[t.userId]) users[t.userId] = { balance: 0 }; users[t.userId].balance += t.rate; res.json({ success: true, message: "অ্যাপ্রুভ করা হয়েছে এবং ব্যালেন্স যোগ হয়েছে!" }); } else if (t && action === 'reject') { t.status = 'Rejected'; res.json({ success: true, message: "রিজেক্ট করা হয়েছে!" }); } else { res.status(400).json({ message: "ত্রুটি হয়েছে" }); } }); app.post('/api/withdraw', (req, res) => { const { userId, method, amount } = req.body; const amt = parseFloat(amount); if (!users[userId] || users[userId].balance < amt || amt < 50) { return res.json({ success: false, message: "অপর্যাপ্ত ব্যালেন্স অথবা সর্বনিম্ন ৫০ টাকা দিতে হবে!" }); } users[userId].balance -= amt; res.json({ success: true, message: `${method}-এর মাধ্যমে ${amt} টাকার উইথড্র রিকোয়েস্ট সফল!` }); }); app.listen(3000, () => console.log('App running on http://localhost:3000'));
