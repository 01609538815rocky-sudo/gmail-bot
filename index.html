import os from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup from telegram.ext import ( ApplicationBuilder, CommandHandler, CallbackQueryHandler, ContextTypes, ) TOKEN = os.getenv("BOT_TOKEN") async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): keyboard = [ [InlineKeyboardButton("🏠 Home", callback_data="home")], [InlineKeyboardButton("ℹ️ About", callback_data="about")], ] await update.message.reply_text( "🤖 স্বাগতম!\n\nবটটি সফলভাবে চালু আছে।", reply_markup=InlineKeyboardMarkup(keyboard), ) async def button_handler(update: Update, context: ContextTypes.DEFAULT_TYPE): query = update.callback_query await query.answer() if query.data == "home": text = "🏠 Home\n\nবটটি বর্তমানে চালু আছে।" elif query.data == "about": text = "ℹ️ About\n\nTelegram Bot" else: text = "অজানা অপশন।" await query.edit_message_text(text) def main(): if not TOKEN: print("ERROR: BOT_TOKEN সেট করা হয়নি।") print("প্রথমে BOT_TOKEN environment variable সেট করুন।") return app = ApplicationBuilder().token(TOKEN).build() app.add_handler(CommandHandler("start", start)) app.add_handler(CallbackQueryHandler(button_handler)) print("✅ Bot is running...") app.run_polling() if __name__ == "__main__": main()
